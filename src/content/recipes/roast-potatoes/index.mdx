  techniqueRationale:
    - stepId: rough-up

  commonPitfalls:
    - stepId: roast

<Step id="parboil">
  Cover the <Qty ref="potato"/> with cold water, add the <Qty ref="salt-water"/>,
  bring to the boil and cook for <Dur step="parboil"/>, until the edges are just
  starting to soften.
</Step>

<Step id="heat-fat">
  Put the <Qty ref="fat"/> into a large roasting tin and heat in the oven at
  <Temp c={220}/> for <Dur step="heat-fat"/>, until shimmering hot.
</Step>

<Step id="rough-up">
  Drain the potatoes well, return them to the dry pan, sprinkle over the
  <Qty ref="flour"/>, and shake hard with the lid on to rough up and coat the edges.
</Step>

<Step id="roast">
  Tip the potatoes carefully into the hot fat, turning to coat, and roast at
  <Temp c={220}/> for <Dur step="roast"/>, turning once or twice, until deep golden
  and crisp all over. Season with the <Qty ref="salt-serve"/> and serve at once.
</Step>
