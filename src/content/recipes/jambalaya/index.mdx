  commonPitfalls:
    - stepId: simmer-jambalaya

<Step id="prep-veg">
  Dice the <Qty ref="onion"/>, <Qty ref="celery"/> and <Qty ref="green-pepper"/>, mince
  the <Qty ref="garlic"/>, slice the <Qty ref="sausage"/>, and cut the
  <Qty ref="chicken"/> into pieces.
</Step>

<Step id="brown-meats">
  Heat the <Qty ref="oil"/> in a large, heavy pot over a medium-high heat and brown the
  sausage and chicken for <Dur step="brown-meats"/>. Lift both out and set aside.
</Step>

<Step id="cook-trinity">
  Add the onion, celery and green pepper to the pot and cook for
  <Dur step="cook-trinity"/>, stirring often, until softened, adding the garlic in the
  last minute.
</Step>

<Step id="add-tomato-rice">
  Stir in the <Qty ref="tomato"/>, <Qty ref="rice"/>, <Qty ref="bay"/>,
  <Qty ref="paprika"/>, <Qty ref="cayenne"/>, <Qty ref="black-pepper"/> and
  <Qty ref="salt"/>, and return the sausage and chicken to the pot.
</Step>

<Step id="simmer-jambalaya">
  Pour in the <Qty ref="water"/>, bring to the boil, then cover and reduce to the
  lowest heat. Cook, undisturbed, for <Dur step="simmer-jambalaya"/>.
</Step>

<Step id="add-shrimp">
  Nestle the <Qty ref="shrimp"/> into the rice, cover again, and cook for
  <Dur step="add-shrimp"/>, until the shrimp are pink and the rice is tender.
  Discard the bay leaves before serving.
</Step>
