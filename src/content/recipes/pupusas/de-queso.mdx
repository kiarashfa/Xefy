  - id: masa-harina

    unit: g

    ingredientRef: water
    form: tap

    unit: ml

    ingredientRef: fine-sea-salt
    form: fine

  - id: oil
    ingredientRef: vegetable-oil
    form: refined
    amount: 15
    unit: ml

  - { id: make-dough, durationMin: 10, type: active, phase: prep }

  - { id: cook-pupusas, durationMin: 16, type: active, phase: cook }
makeAhead:
  aheadInstructions: The dough is best made and used the same day.

<Step id="make-dough">
  Combine the <Qty ref="masa-harina"/> and <Qty ref="dough-salt"/>, then work
  in the <Qty ref="dough-water"/> to form a soft, smooth dough. This takes
  about <Dur step="make-dough"/>.
</Step>

<Step id="rest-dough">
  Cover the dough and leave it to rest for <Dur step="rest-dough"/>.
</Step>

<Step id="shape-and-fill">
  Divide the dough into as many equal balls as pupusas are being made.
  Flatten each into a cup, fill generously with the <Qty ref="cheese"/>,
  then draw the edges up and pinch closed around it. Flatten each filled
  ball gently between the palms into a thick disc, taking care not to let
  the cheese break through. This takes about <Dur step="shape-and-fill"/> in
  total.
</Step>

<Step id="cook-pupusas">
  Cook the discs on a lightly oiled griddle with the <Qty ref="oil"/> over a
  moderate heat for <Dur step="cook-pupusas"/> in total, turning once or
  twice, until both sides are browned in spots, the dough is cooked through
  and the cheese has melted.
</Step>
